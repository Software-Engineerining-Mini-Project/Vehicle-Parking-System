6# Default 6 x 8 facility layout (OR-01). Numbers are slot IDs; the numbering is the one 

7# the Test Plan uses: S01-S10 Car, S11-S14 Two-Wheeler, S15-S16 Accessible, S17-S20 EV. 

8# # wall . road N entrance X exit 

9LAYOUT = [ 

10 "# 01 02 03 04 05 06 #", 

11 "N . . . . . . .", 

12 "# 11 12 13 14 # # .", 

13 "# 20 19 18 17 # # .", 

14 "# . . . . . . X", 

15 "# 16 15 07 08 09 10 #", 

16] 

17 

18OTHER_CELLS = {"#": "WALL", ".": "ROAD", "N": "ENTRANCE", "X": "EXIT"}

20 

21def slot_category(number): 

22 """Return the default category of a slot number.""" 

23 if number <= 10: 

24 return "CAR" 

25 if number <= 14: 

26 return "TWO_WHEELER" 

27 if number <= 16: 

28 return "ACCESSIBLE" 

29 return "EV" 

30

32DEFAULT_SETTINGS = { 

33 "rate_two_wheeler": "20", 

34 "rate_car": "40", 

35 "rate_ev": "50", 

36 "grace_minutes": "5", 

37 "overstay_hours": "24",

42def seed_demo_data(db): 

43 """Insert the default layout, slots, settings and the two demo accounts.""" 

44 for row_no, line in enumerate(LAYOUT): 

45 for col_no, token in enumerate(line.split()): 

46 if token.isdigit(): 

47 slot_id = f"S{token}" 

48 db.execute( 

49 "INSERT INTO slots (slot_id, category) VALUES (?, ?)", 

50 (slot_id, slot_category(int(token))), 

51 ) 

52 cell = ("SLOT", slot_id) 

53 else: 

54 cell = (OTHER_CELLS[token], None) 

55 db.execute( 

56 "INSERT INTO grid_cells (row_no, col_no, cell_type, slot_id)" 

57 " VALUES (?, ?, ?, ?)", 

58 (row_no, col_no, *cell), 

59 ) 

60 

61 db.executemany( 

62 "INSERT INTO settings (key, value) VALUES (?, ?)", DEFAULT_SETTINGS.items() 

63 ) 

64 

65 # Demo passwords can be overridden through environment variables. 

66 accounts = [

70 for username, password, role in accounts: 

71 db.execute(
